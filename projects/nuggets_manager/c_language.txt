••• C: Read the contents of a file
void read_file(const char *file_name) {
  FILE *fp;
  size_t length;
  char *buffer;

  fp = fopen(file_name, "rb");

  if (fp) {
    fseek(fp, 0, SEEK_END);
    length = ftell(fp);
    fseek(fp, 0, SEEK_SET);

    buffer = (char *)malloc(length + 1);
    if (buffer) {
      size_t read_length = fread(buffer, 1, length, fp);
      buffer[read_length] = '\0';  // Null-terminate the buffer

      if (read_length != length) {
        perror("Failed to read the entire file");
        free(buffer);
        fclose(fp);
        exit(EXIT_FAILURE);
      }

      fprintf(stdout, "%s", buffer);
      free(buffer);
    } else {
      perror("Failed to allocate memory for reading file");
      fclose(fp);
      exit(EXIT_FAILURE);
    }
    fclose(fp);
  } else {
    perror("Failed to open the file");
    exit(EXIT_FAILURE);
  }
}

#include <stdio.h>

#include <string.h>

int main(void) {
    char *str;

    // Initial memory allocation. Always allocate one byte more for the null terminator with strings.
    str = malloc(15 * sizeof(char)); // No need to cast the result of malloc in C.
    if (str == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        return 1; // Return an error code if memory allocation fails.
    }
    strcpy(str, "tutorialspoint");
    printf("String = %s, Address = %p\n", str, str);

    // Reallocating memory
    char *temp = realloc(str, 25 * sizeof(char)); // Use a temporary pointer when reallocating.
    if (temp == NULL) {
        fprintf(stderr, "Memory reallocation failed\n");
        free(str); // Free the original block of memory if realloc fails.
        return 1; // Return an error code if memory reallocation fails.
    }
    str = temp; // Only update str if realloc was successful.
    strcat(str, ".com");
    printf("String = %s, Address = %p\n", str, str);

    free(str); // Free the allocated memory.
    return 0;
}

int remove_last(node_t *head) {
    int retval = 0;
    /* if there is only one item in the list, remove it */
    if (head->next == NULL) {
        retval = head->val;
        free(head);
        return retval;
    }

    /* get to the second to last node in the list */
    node_t * current = head;
    while (current->next->next != NULL) {
        current = current->next;
    }

    /* now current points to the second to last item of the list, so let's remove current->next */
    retval = current->next->val;
    free(current->next);
    current->next = NULL;
    return retval;

}

int sum(int args, ...) {
  va_list list; 
  int sum = 0; 

  // Early return if no arguments are passed
  if (args == 0)
    return 0;

  printf("Num of argument passed: %d\n", args);

  // Initialize 'list' to retrieve the additional arguments
  va_start(list, args);

  // Retrieve the next argument as an integer
  for (int i = 0; i < args; i++) {
    sum += va_arg(list, int);
  }
  va_end(list); // Clean up the variable argument list

  return sum; // Return the sum of the arguments
}
