from collections import defaultdict


def word_count(text):
    """ Count the frequency of words in a text """

    freq = defaultdict(int)

    for word in text.lower().split():
        freq[word] += 1

        freq_sorted = sorted(freq.items(), key=lambda x: x[1], reverse=True)

    return freq_sorted

if __name__ == "__main__":
    print(word_count("This is this a test test message"))

••• Python short scripts -  Emulate switch/case

def dispatch_if(operator: str, x: int, y: int) -> int:
    """Simulate switch/case statements"""

    if operator == "add":
        return x + y
    elif operator == "sub":
        return x - y
    elif operator == "mul":
        return x * y
    elif operator == "div":
        return x / y
    else:
        return None


def dispatch_dict(operator: str, x: int, y: int) -> int:
    """Simulate switch/case statements"""

    return {
        "add": lambda: x + y,
        "sub": lambda: x - y,
        "mul": lambda: x * y,
        "div": lambda: x / y,
    }.get(operator, lambda: None)()

print(dispatch_if("add", 23, 43))
print(dispatch_dict("add", 23, 43))

from itertools import filterfalse

def first_try(text: str, min_length: int) -> int:

    words = text.split(" ")
    return len([word for word in words if len(word) > min_length])


def second_try(text: str, min_length: int) -> int:

    words = text.split(" ")
    return len(list(filterfalse(lambda x: len(x) < min_length, words)))

••• Python short scripts -  Binary search
def binary_search(target: int, array: list):
    if not array:
        return False

    sorted_array = sorted(array)
    low, high = 0, len(sorted_array) - 1

    while low <= high:
        mid = (low + high) // 2
        current = sorted_array[mid]
        if current == target:
            return True
        if target > current:
        # -->
            low = mid + 1
        else:
        # <--
            high = mid - 1
    return False


array = [111, 2, 3, 40, 5, 6, 7, 88, 9, 10]
print(binary_search(111, array))

def count_names(names: list[list[str]], target: str) -> int:
    if not names:
        return 0
    return sum(target == name for sublist in names for name in sublist)

                                                     "Natalia"]]
