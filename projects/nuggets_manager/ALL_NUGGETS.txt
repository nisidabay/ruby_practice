local function find_files(dir, patterns)
  local cmd = 'find "' .. dir .. '" -type f '
  for _, pattern in ipairs(patterns) do
    cmd = cmd .. '-o -iname "' .. pattern .. '" '
  end
  local handle = io.popen(cmd)
  local output = handle:read("*a")
  handle:close()
  local files = {}
  for file in output:gmatch("[^\n]+") do
    table.insert(files, file)
  end
  return files
end

••• Argparse - Setting the Name or Flags of the Arguments

There are basically two different types of arguments that you can add to your 
command line interface:

Positional arguments
Optional arguments

Positional arguments are the ones your command needs to operate.
Optional arguments start with - or --, while positional arguments don’t.

To add an optional argument, you just need to call .add_argument() again and 
name the new argument with a starting -.

my_parser = argparse.ArgumentParser(description='List the content of a folder')

my_parser.add_argument('Path',
                       metavar='path',
                       type=str,
                       help='the path to list')
my_parser.add_argument('-l',
                       '--long',
                       action='store_true',
                       help='enable the long listing format')
args = my_parser.parse_args()

••• Argparse - Setting the Action to Be Taken for an Argument

There are several actions that are already defined and ready to be used. 

store 
------
stores the input value to the Namespace object. (This is the default action.)

store_const 
------------
stores a constant value when the corresponding optional arguments are specified.

store_true 
-----------
stores the Boolean value True when the corresponding optional argument is 
specified and stores a False elsewhere.

store_false 
------------
stores the Boolean value False when the corresponding optional argument is 
specified and stores True elsewhere.

append 
-------
stores a list, appending a value to the list each time the option is provided.

append_const 
-------------
stores a list appending a constant value to the list each time the option is 
provided.

count 
------
stores an int that is equal to the times the option has been provided.

help 
-----
shows a help text and exits.

version 
--------
shows the version of the program and exits.

••• Argparse - Setting the Number of Values That Should Be Consumed by the Option

The parser, by default, assumes that you’ll consume a single parameter for each 
argument, but you can modify this default behavior by specifying a different 
number of values with the nargs keyword.

You can specify the number of arguments: 

my_parser = argparse.ArgumentParser()
my_parser.add_argument('--input', action='store', type=int, nargs=3)

args = my_parser.parse_args()

print(args.input)
Now, the program accepts three values for the --input parameter:

$ python nargs_example.py --input 42
usage: nargs_example.py [-h] [--input INPUT INPUT INPUT]
nargs_example.py: error: argument --input: expected 3 arguments

void read_file(const char *file_name) {
  FILE *fp;
  size_t length;
  char *buffer;

  fp = fopen(file_name, "rb");

  if (fp) {
    fseek(fp, 0, SEEK_END);
    length = ftell(fp);
    fseek(fp, 0, SEEK_SET);

    buffer = (char *)malloc(length + 1);
    if (buffer) {
      size_t read_length = fread(buffer, 1, length, fp);
      buffer[read_length] = '\0';  // Null-terminate the buffer

      if (read_length != length) {
        perror("Failed to read the entire file");
        free(buffer);
        fclose(fp);
        exit(EXIT_FAILURE);
      }

      fprintf(stdout, "%s", buffer);
      free(buffer);
    } else {
      perror("Failed to allocate memory for reading file");
      fclose(fp);
      exit(EXIT_FAILURE);
    }
    fclose(fp);
  } else {
    perror("Failed to open the file");
    exit(EXIT_FAILURE);
  }
}

        retval = head->val;
        free(head);
        return retval;
    }

    node_t * current = head;
    while (current->next->next != NULL) {
        current = current->next;
    }

    /* now current points to the second to last item of the list, so let's remove current->next */
    retval = current->next->val;

local variables take up memory only while the function they are defined in is
executing. Hence, use of local variables reduces memory usage. More important,
however, is that the use of local variables greatly decreases the chance of
unwanted interactions between different parts of the program, hence lessening
program bugs and following the principles of structured programming.

single element.

#!/usr/bin/bash
# Author: Carlos Lacaci Moya
# Description: Send files/folders over SCP to a series of hosts
# This is a small wrapper of "remote_transfer.sh"
# Date: mié 15 dic 2021 21:23:31 CET
# Dependencies: remote_transfer.sh

# Debugging setup for bash
set -euo pipefail
################################################################################ 
# GLOBAL VARIABLES
################################################################################

# Ansi color code global variables
expand_bg="\e[K"
blue_bg="\e[0;104m${expand_bg}"
red_bg="\e[0;101m${expand_bg}"
green_bg="\e[0;42m${expand_bg}"
green="\e[0;32m\033[1m"
red="\e[0;31m\033[1m"
turquoise="\e[0;36m\033[1m"
gray="\e[0;37m\033[1m"
yellow="\e[1;93m\033[1m"
reset="\033[0m\e[0m"

DEPENDENCIES=(remote_transfer.sh)
OK='✔'
FAIL='✘'
declare -a favorite_hosts
################################################################################

favorite_hosts=(Tecra.home BlackManjaro.home iMacEndeavourOs.home pop-os.home)

function check_program_dependencies(){
# Check for necessary binaries

    for prog in "${DEPENDENCIES[@]}"; do
    
        # Check for remote_transfer.sh in HOME
        [[ -f "$HOME/bin/$prog" ]] > /dev/null 2>&1
        if [ $? -ne 0 ];then
            echo -e "\t[-] $prog -> ${red}$FAIL${reset}"
            exit 1
        fi

    done
}

function get_ip_from_name(){
# Get the ip from the favorite_hosts

name=$1
eth=$(nslookup $name | awk -F ":" '$1 == "Address" { print $2 }' | sed -ne '/#/!p')

if [[ -z $eth ]];then
    echo -e "${red}[-] Unknown host!${reset}"
    exit 1
fi

# Return the value
echo $eth
}

function transfer_files(){
# Transfer files to remote hosts

    # This computer
    myself=$(hostname)

    for t in "${favorite_hosts[@]}"; do

        barename=${t%.home}
        # Skip myself from the favorite_hosts
        if [ $barename == $myself ];then
            continue
        fi

        # Get the name of the host
        name=$(get_ip_from_name "$t")

        # Is online?
        value=$(ping -c1 "$name" > /dev/null 2>&1 && echo 0 || echo 1)

        if [[ $value -eq 0 ]];then
            remote_transfer.sh -i "$name" -d "$host_folder" -f "$files" 

        elif [[ $value -eq 1 ]];then
            echo -e "[!] Host [${red}$t${reset}] unreachable ${red}$FAIL${reset}\n"
            continue
        fi
    done
}

function show_help(){
# Usage

  echo -ne "${yellow}Transfer files favorite hosts\n${reset}"
  echo
  echo -ne "${gray}Usage: transfer2fav.sh [OPTIONS]\n"

  echo -ne "\ttransfer.sh -d . -f \"movie_1.mkv\"\n"
  echo -ne "\ttransfer.sh -d Downloads -f \"Movies\"\n"
  echo -ne "\ttransfer.sh -d . -f \"Movies\"\n"

  echo -ne "${yellow}@Carlos Lacaci Moya - 2021 ;)\n${reset}"
  exit 0
}

>>> {'name': 'Jane', 'job': 'Python Developer'}

    DivMod = namedtuple("DivMod", "quotient remainder")

••• Concurrency: The ability of a program to deal with multiple tasks seemingly
    simultaneously. This doesn't necessarily mean they are executing at the
    exact same moment.

••• Parallelism: A form of concurrency where multiple tasks are executed at the
    exact same time, typically on multiple CPU cores.

••• Thread: A lightweight unit of execution within a process. Multiple threads
    can run concurrently within the same process and share the same memory
    space.

••• Process: An independent unit of execution with its own memory space.
    Processes are typically heavier than threads.

••• I/O-bound Task: A task where the primary bottleneck is waiting for
    input/output operations to complete (e.g., reading files, network requests).

••• CPU-bound Task: A task where the primary bottleneck is the processing power
    of the CPU (e.g., complex calculations, data encoding).

••• Global Interpreter Lock (GIL): A mutex (lock) in CPython that prevents
    multiple native threads from executing Python bytecode at the same time
    within a single process.

••• Thread Pool: A collection of worker threads that are created and managed to
    execute tasks concurrently. This avoids the overhead of creating new threads
    for each task.

••• Process Pool: A collection of worker processes that are created and managed
    to execute tasks in parallel.

••• Executor: An object that manages the execution of asynchronous tasks. The
    concurrent.futures module provides ThreadPoolExecutor and
    ProcessPoolExecutor.

••• Future Object: An object returned by an executor's submit() method. It
    represents the result of an asynchronous operation and provides methods to
    check its status, retrieve the result, or handle exceptions.

••• Pickling (Serialization): The process of converting Python objects into a
    byte stream so that they can be stored or transmitted across a network or
    between processes.

••• Context Manager: An object that defines methods to be executed at the start
    and end of a controlled block (e.g., using the with statement). This is
    often used for resource management, like ensuring a thread or process pool
    is properly closed.

def dispatch_if(operator: str, x: int, y: int) -> int:
    """Simulate switch/case statements"""

    if operator == "add":
        return x + y
    elif operator == "sub":
        return x - y
    elif operator == "mul":
        return x * y
    elif operator == "div":
        return x / y
    else:
        return None


def dispatch_dict(operator: str, x: int, y: int) -> int:
    """Simulate switch/case statements"""

    return {
        "add": lambda: x + y,
        "sub": lambda: x - y,
        "mul": lambda: x * y,
        "div": lambda: x / y,
    }.get(operator, lambda: None)()

print(dispatch_if("add", 23, 43))

••• Python short scripts -  Binary search
def binary_search(target: int, array: list):
    if not array:
        return False

    sorted_array = sorted(array)
    low, high = 0, len(sorted_array) - 1

    while low <= high:
        mid = (low + high) // 2
        current = sorted_array[mid]
        if current == target:
            return True
        if target > current:
        # -->
            low = mid + 1
        else:
        # <--
            high = mid - 1
    return False


array = [111, 2, 3, 40, 5, 6, 7, 88, 9, 10]
print(binary_search(111, array))

••• Rsync

The rsync commands uses compression and decompression when receiving and
sending files, this means it will consume very less bandwidth. Rsync only
updates the differences between the source and destination as it uses the
remote update protocol. It also supports owners, groups, links, permissions,
and devices.

Local to local transfer or sync
-------------------------------
rsync {option} {source} {destination}

Local to remote transfer or sync
--------------------------------
rsync {option} {source} {user@host:destination}

Remote to local transfer or sync
--------------------------------
rsync {option} {user@host:destination} {destination}

••• Rsync -  Copy files locally

rsync -a {filename} {destination}
rsync -a source.zip /home/nisidabay/Documents

If you don’t add a name for the file in the destination, it will copy the same 
name as the source file. To change the name of the destination file, use:

rsync –a {filename} {destination/filename}

••• Rsync -  Copy the contents of a directory to the destination

rsync -av {source}/ {destination}

••• Rsync -  Copy the whole directory to the destination

rsync -av {source} {destination}

••• Rsync -  Show progress

rsync –rv –progress {source} {destination}

••• Rsync -  Ignore files

rsync -vr --exclude '{condition}' {source} {destination}

rsync -vr --exclude '*.tmp' {source} {destination}

••• Rsync -  List file and directories

rsync {filename}

••• Rsync -  Copy files from a local to a remote machine

For moving files to a remote machine, you will need to install openssh-server.

rsync –a {source} {remote username@remote IP:/destination}
rsync -a /home/nisidabay/Documents nisidabay@192.168.1.38:/home/nisidabay

••• Rsync -  Copy files from a remote to a local machine

rsync –a {remote source} {local username@local IP:/destination}
rsync -a /home/nisidabay/doc nisidabay@192.168.1.69:/home/nisidabay

••• Rsync -  Add current date to the destination file

rsync -av /home/test/Desktop/Linux /home/test/Desktop/rsync$(date +\\%Y-\\%m)

••• Rsync -  View differences between the source and destination files

rsync -avzi {source}/ {destination}/

The parameter “f” means file. Other parameters include:

t: change in timestamp

d: change in destination

s: change in file size

••• Rsync -  Limit bandwidth when transferring files

To limit the bandwidth while transferring files, use:

rsync -vr --bwlimit=1000 {source}/* {destination}

••• Python - tkinter - Center widgets with grid
The pack() geometry manager can be used to center widgets automatically by
using options like expand=True and fill=tk.BOTH. 

However, when using the grid() geometry manager, you need to manually configure
the rows and columns to ensure that widgets are centered. 

This is typically done using grid_rowconfigure and grid_columnconfigure to set
the weight of rows and columns, allowing them to expand and center the widgets
within the grid.

# Configure the grid to center the widgets
self.root.grid_rowconfigure(0, weight=1)
self.root.grid_columnconfigure(0, weight=1)

••• Python - tkinter - Pack and Grid
They use different logic to lay out widgets. For this reason they cannot be 
combined inside the same parent widget. Once one widget is inserted using pack
or grid, all other widgets must use the same geometry manager. You can however
pack widgets into one frame, grid widgets into another, then pack/grid both of
those Frame widgets into the same parent widget.

• Convert lower case to upper case  

Converting lower case to upper case is one of the typical use cases 
of the tr command. [:lower:] matches all lower case characters and [:upper:] 
matches all uppercase characters.

echo 'Linuxize' | tr '[:lower:]' '[:upper:]'
>>> output: LINUXIZE

Instead of character classes, you can also use ranges
echo 'Linuxize' | tr 'a-z' 'A-Z'

>>> output: LINUXIZE

f(range(1, 3))

Mapping describes a dict-like object (with "__getitem__") that we won't

def f(my_mapping: Mapping[int, str]) -> list[int]:
    my_mapping[5] = 'maybe'  - if we try this, mypy will throw an error...
    return list(my_mapping.keys())

def f(my_mapping: MutableMapping[int, str]) -> set[str]:

The groupby iterator will return consecutive keys and groups from your
iterable.

from itertools import groupby

zipcodes = [("sergio", 11100), ("dani", 11100), ("carlos", 28039),
            ("alicia", 28039), ("clara", 28040)]

sorted_zipcodes = (sorted(zipcodes))

for _, group in groupby(sorted_zipcodes, lambda name: name[0]):
    for name, zipcode in group:
        print(f"name:{name} zipcode:{zipcode}")

••• xargs
Lee un grupo de argumentos de la entrada standard (stdin) y después ejecuta un
comando Unix con cada grupo de argumentos hasta que los usa todos.

••• xargs ejecuta salida del comando que precede. 
find . -type f -print | xargs grep -l WARNING | xargs pr | lpr

••• xargs produce "echo" como comando por defecto.
xargs "Hello" (Ctr-d) -> Hello

••• xargs - opciones más usadas
(t) Salida detallada (verbose)
(n) Establece el número máximo de argumentos que xargs pasará a cada comando.
Por defecto 5000.

ls | xargs -n 1 grep -in "python|bash" *.py *.sh

(p) Prompt. Espera confirmación antes de ejecutar el comando.

echo "one two three" | xargs -p touch

(L) Número de líneas a procesar sin espacios en blanco.

xargs -L 1 grep -in "python|bash" *.py *.sh

(0) Ignora los espacios en blanco o saltos de línea.

find . -name "foo*" -print0 | xargs -O vim

(a) Lee los argumentos de un archivo en vez de la entrada standard(stdin).

xargs -a input.txt ls -l

(I) Es un comodín o cadena de reemplazo que contiene el archivo a pasar a otro
comando. Se puede usar % en lugar de {}.

cat directories.txt | xargs -I % sh -c "echo creando %; mkdir %"

••• xargs crear directorios de un listado
cat directories.txt | xargs -I % sh -c "echo creando %; mkdir %"

••• xargs eliminar archivos en directorios anidados.
find . -name "*.png" -type f -print0 | xargs -0 -I {} rm -v -rf "{}"

••• xargs eliminar todos los archivos excepto un tipo de archivo.
find . -type f -not -name "*.sh" -print0 | xargs -0 -I {} rm -v {}

••• xargs comprimir un archivo
find . -name "*.page" -type f -print0 | xargs -0 tar cvzf page_files.tar.gz

••• xargs Crear 5 archivos
seq 5 | xargs -n 1 -I{} touch {}.txt

••• xargs crear archivos desde un fichero
cat filenames.txt | xargs -n 1 touch
cat filenames.txt | xargs -I {} -n 1 touch {}.txt

••• xargs borrar archivos
find . -type f -name "*.txt" -print | xargs -n 1 -t rm 

••• xargs borrar archivos y preguntar
find . -maxdepth 1 -type f -name "[0-9].txt" | xargs -p -t rm

••• xargs borrar archivos txt del primer nivel
find . -maxdepth 1 -type f -name "*.txt" | xargs -L1 -I {} rm {}

••• xargs Copiar archivos txt a un directorio
ls *.txt | xargs -n 1 -I {} cp -v {} ./temp

••• xargs mover ciertos archivos a un directorio
find . -type f -maxdepth 1 -not -name "*.sh" | xargs -n 1 -I{} mv {} dest/{}

••• xargs renombrar extensión de archivos txt
ls | cut -d. -f1 | xargs -I {} mv {}.txt {}.text
