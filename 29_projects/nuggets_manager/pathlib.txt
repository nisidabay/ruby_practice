••• Python - pathlib -  Remove a directory along with its contents
from pathlib import Path

def remove_all(root: Path):
    for path in root.iterdir():
        if path.is_file():
            print(f'Deleting the file: {path}')
            path.unlink()
        else:
            remove_all(path)
        print(f'Deleting the empty dir: {root}')
        root.rmdir()

path = Path('/home/nisidabay/Downloads')
remove_all(path)

••• Python - pathlib -  Implement a dir tree

from pathlib import Path

def tree(directory) -> None:
    print(f"+ {directory}")
    for path in sorted(directory.rglob("*")):
        depth = len(path.relative_to(directory).parts)
        spacer = "   " * depth
        print(f"{spacer}+ {path.name}")


if __name__ == "__main__":
    tree(Path.cwd())
