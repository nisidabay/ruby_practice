••• zenity -  Progress bar 
(
echo 25
echo "hashtag Setting up..."
sleep 1

echo 30
echo "hashtag Reading files..."
sleep 1

echo 70
echo "hashtag Creating content..."
sleep 1

echo 100
echo "hashtag Done..."
) | zenity --title "Progress bar example" --progress --auto-kill

(echo "40";sleep 1; echo "60";sleep 1; echo "100";sleep 1) \
| zenity --title="auto-kill" --progress --text="Working hard or hardly working" \
    --percentage=25 --auto-kill

(echo "40";sleep 1; echo "60";sleep 1; echo "100";sleep 1) \
| zenity --title="auto-close" --progress --text="Working hard or hardly working" \
    --percentage=25 --auto-close
