#include <stdio.h>

int main(void) {
    char answer;
    do {
        printf("Continue? (y/n): ");
        answer = getchar();
        while (getchar() != '\n'); // Consume the newline character
    } while (answer != 'y' && answer != 'n');

    return 0;
}

•••C: Read the contents of a file

#include <stdio.h>
#include <stdlib.h>

void read_file(const char *filename) {
    FILE *fp;
    char *buffer;
    long length;

    fp = fopen(filename, "rb"); // Open the file in binary mode
    if (fp) {
        fseek(fp, 0, SEEK_END);
        length = ftell(fp);
        fseek(fp, 0, SEEK_SET);
        buffer = malloc(length + 1);
        if (buffer) {
            fread(buffer, 1, length, fp);
            buffer[length] = '\0'; // Null-terminate the string
        }
        fclose(fp);
        printf("%s\n", buffer);
        free(buffer);
    } else {
        fprintf(stderr, "File not found: %s\n", filename);
    }
}

#include <stdlib.h>

int main(void) {
    FILE *fp;
    char *filename = "output.txt";
    char *account = "user123";
    char *password = "pass123";

    if ((fp = fopen(filename, "a")) != NULL) {
        fprintf(fp, "account: %s - password: %s\n", account, password);
        fclose(fp); // Correct use of fclose
    } else {
        perror("Error opening file");
        return EXIT_FAILURE;
    }

    return 0;
}

•••c_language_tips: Stack

The stack is a structured region of memory that stores variables created by each
function (including the main function) during its execution. The stack operates
in a last-in, first-out manner, meaning the most recently allocated block will
be the first to be freed.

•••C: Use of feof 

#include <stdio.h>

int main() {
  FILE *fp;
  int c;

  fp = fopen("file.txt", "r");
  if (fp == NULL) {
    perror("Error in opening file");
    return (-1);
  }

  while (1) {
    c = fgetc(fp);
    if (feof(fp)) {
      break;
    }
    printf("%c", c);
  }
  fclose(fp);

  return (0);
}
