••• Bash - Arithmetic evaluation
# Prints odd numbers from 1 to 13 using arithmetic evaluation.
num=1
while [[ $num -le 20 ]]; do
    if (( num % 2 == 0 )); then
        num=$((num + 1))
        continue
    fi
    if (( num >= 15 )); then
        break
    fi
    echo $num
    num=$((num + 1))
done

••• Bash - Usage of trap
# Cleans up .txt files if the script is interrupted or killed.
function cleanup() {
    echo
    echo "Script killed."
    echo "Cleaning files ..."
    find . -maxdepth 1 -type f -name "*.txt" -delete
    exit 0
}

# Trap termination signals and call cleanup.
trap cleanup SIGTERM
trap cleanup SIGINT

PROGRAM=$(basename "$0")
echo "[$PROGRAM] PID: $$"

# Creates dummy text files and writes dummy content.
for i in $(seq 1 10); do
    echo "Creating file $i.txt"
    touch "$i.txt"
    echo "Dummy text" >> "$i.txt"
    sleep 2
done

••• Bash - Spinner for background process
# Displays a spinner while a background process runs.
sleep 5 &
pid=$!

frames="/ | \ -"

while kill -0 "$pid" 2>/dev/null; do
    for frame in $frames; do
        printf "\r$frame Loading..."
        sleep 0.5
    done
done
printf "\n"

••• Bash - Error handling in mkdir function
# Creates a directory with optional permissions and group ownership.
set -euo pipefail

function _mkdir() {
    local directory=${1:-"/tmp"}
    local permission=${2:-0775}

    if [[ $# -eq 0 ]]; then
        echo "$0: [directory] [octal permissions]"
        return
    fi

    if [[ ! -d "$directory" ]]; then
        mkdir -m "$permission" -p "$directory"
        chgrp -R users "$directory"
    fi
}
_mkdir "$@"
